Why Fiducia

Security and correctness, built in.

Fiducia closes the trust gap between the formally verified seL4 kernel and the applications built on top. It catches entire classes of bugs at compile time that would be invisible at runtime.

The language

Process-Oriented Programming

Each protection domain is a CSP-style process that orchestrates channel communication and computation events, calls into C, with guarded choice. Clear, composable and verifiable.

Secure by Design

A two-level security type system (Low/High) statically enforces that sensitive data never leaks to lower-security domains. No runtime checks needed.

Deadlock-Free by Design

Every well-formed system, with acyclic communication dependencies and guarded choices that partition the state space, is deadlock-free and deterministic.

The verification story

Design by Contract

Computation stays in C or Pancake. Its ACSL or Viper contracts are proved with Frama-C or Viper, and each contract becomes the event's specification in the system model.

Continuous Verification

Every build is also a verification run: fic derives a UPPAAL model of the whole system directly from your program (autoformalisation), so the model never drifts from the code that ships.

Proved Infrastructure

The system runs on seL4 Microkit, a formally verified microkernel. CoreFi's semantics, deadlock freedom and determinism are mechanised in Lean 4, and certified compilation of fic is in progress.

How it works

Code, formalise, verify, repair. Then build.

Every check that fails produces a counterexample, and the counterexample drives the next repair. Only a system where every check holds is compiled to an image.

Code + contract. The system in .fis, each protection domain in .fi, computation in C or Pancake under ACSL or Viper contracts.
Autoformalisation. fic derives a UPPAAL model of the whole system from the program itself, so the model and the image come from the same source.
Code verification. Each call into foreign code carries a contract on its implementation, proved with Frama-C or Viper.
System verification. fic checks information flow and channels; the UPPAAL model checker checks the generated model.
Counterexample, repair & synthesis. A failed check yields a concrete trace, used to fix the code or contract, or synthesise the missing piece.
Executable image. What holds is compiled to a bootable seL4 Microkit image.

// Data diode: eth_outer → dd → eth_inner, one-way Low → High.

// A frame crosses the diode by copy, as the channels' value; `data` holds
// a virtio-net buffer (`VIRTIO_NET_BUF_SIZE`).
struct Frame {
    len: uint16;
    // The NIC the frame came in on (ORIGIN_*), which the model tracks for
    // secrecy; the bytes stay in the C.
    origin: uint8;
    data: uint8[2048];
}

// A frame's origin: the outer (Low) or inner (High) network. Neither is 0,
// so a zeroed frame has no origin.
const ORIGIN_OUTER: uint8 = 1;
const ORIGIN_INNER: uint8 = 2;

// The channels' rings, mapped into both endpoints of their channel.
Low MemoryRegion eth_outer_output(0x200000, 0x200000);
High MemoryRegion eth_inner_input(0x200000, 0x200000);

// One 4 KiB page of QEMU virt's virtio-mmio slots 24..31, where both NICs
// sit; each eth PD binds a different device in it.
MemoryRegion virtio_window(0x1000, 0x1000, 0xa003000);

// Virtio rings + buffers; physically contiguous, paddr patched by the tool.
Low MemoryRegion outer_dma(0x200000, 0x200000);
High MemoryRegion inner_dma(0x200000, 0x200000);

// IRQ = 48 + mmio slot. QEMU puts the first -device in slot 31 (IRQ 79),
// the second in slot 30 (IRQ 78).
ProtectionDomain eth_outer(99, 79);
ProtectionDomain dd(150);
ProtectionDomain eth_inner(199, 78);

Chan o2d = eth_outer --> dd via eth_outer_output<Frame>;
Chan d2i = dd --> eth_inner via eth_inner_input<Frame>;

System = eth_outer ||| dd ||| eth_inner;

    poll -> Skip;

static uint8_t cur[VIRTIO_NET_BUF_SIZE];
static uint16_t cur_len;
static int cur_kind;

static uint8_t scratch[VIRTIO_NET_BUF_SIZE];

int64_t eth_setup(int nth, uintptr_t dma_paddr)
{
    return virtio_net_up(vwin_vaddr, (unsigned)nth, dma_vaddr, dma_paddr);
}

void eth_clear_irq(void)
{
    virtio_net_ack();
}

int64_t eth_poll(void)
{
    if (!virtio_net_rx(cur, &cur_len)) {
        cur_kind = 0;
        return 0;
    }
    cur_kind = 0;
    if (cur_len >= sizeof(struct eth_header)) {
        struct eth_header *hdr = (struct eth_header *)cur;
        if (!(mac_match(hdr->dest_mac, fic_eth_mac) ||
              mac_match(hdr->dest_mac, (const uint8_t *)"\xff\xff\xff\xff\xff\xff"))) {
            cur_kind = 1;
        } else if (swap16(hdr->ethertype) == ETHERTYPE_ARP) {
            cur_kind = 2;
        } else if (swap16(hdr->ethertype) == ETHERTYPE_IPV4) {
            cur_kind = 3;
        }
    }
    return 1;
}

int64_t eth_pkt_kind(void)
{
    /* cur_kind is one of 0..3; the test lets WP see it. */
    if (cur_kind < 0 || cur_kind > 3) {
        return 0;
    }
    return cur_kind;
}

struct Frame eth_frame(uint8_t origin)
{
    struct Frame f;
    f.len = cur_len;
    f.origin = origin;
    memcpy(f.data, cur, cur_len);
    return f;
}

uint16_t eth_make_reply(void)
{
    uint16_t len = cur_len;
    struct eth_header *hdr = (struct eth_header *)cur;
    memcpy(scratch, cur, len);

    struct eth_header *reply = (struct eth_header *)scratch;
    set_mac(reply->dest_mac, hdr->src_mac);
    set_mac(reply->src_mac, fic_eth_mac);

    if (swap16(reply->ethertype) == ETHERTYPE_ARP) {
        struct arp *a = (struct arp *)&reply->payload[0];
        a->oper = swap16(2);
        set_mac(a->tha, a->sha);
        set_ip(a->tpa, a->spa);
        set_mac(a->sha, fic_eth_mac);
        set_ip(a->spa, fic_eth_ip);
    } else {
        struct ip *ip_hdr = (struct ip *)&reply->payload[0];
        uint8_t header_len = (uint8_t)((ip_hdr->ver_ihl & 0xf) * 4);
        if (ip_hdr->protocol == 1) {
            struct icmp *icmp_hdr = (struct icmp *)(&reply->payload[header_len]);
            set_ip(ip_hdr->dest_address, ip_hdr->source_address);
            set_ip(ip_hdr->source_address, fic_eth_ip);
            icmp_hdr->type = 0; /* echo reply */
            icmp_hdr->checksum = 0;
            icmp_hdr->checksum =
                cksum((uint8_t *)icmp_hdr, (size_t)(swap16(ip_hdr->len) - header_len));
        }
    }
    return len;
}

void eth_send_frame(uint16_t len)
{
    virtio_net_tx(scratch, len);
}

void eth_send_frame_value(struct Frame f)
{
    virtio_net_tx(f.data, f.len);
}

const uint8_t fic_eth_mac[6] = {0x00, 0x16, 0x3e, 0x12, 0x34, 0x56};

uint16_t swap16(uint16_t v)
{
    return (uint16_t)(((v & 0xff) << 8) | (v >> 8));
}

int mac_match(const uint8_t *a, const uint8_t *b)
{
    for (unsigned i = 0; i < 6; i++) {
        if (a[i] != b[i]) {
            return 0;
        }
    }
    return 1;
}

void set_mac(uint8_t *dst, const uint8_t *src)
{
    for (unsigned i = 0; i < 6; i++) {
        dst[i] = src[i];
    }
}

void set_ip(uint8_t *dst, const uint8_t *src)
{
    for (unsigned i = 0; i < 4; i++) {
        dst[i] = src[i];
    }
}

/* RFC 1071 internet checksum. */
uint16_t cksum(const uint8_t *d, size_t len)
{
    uint32_t sum = 0;
    while (len > 1) {
        sum += (uint32_t)((d[0] << 8) | d[1]);
        d += 2;
        len -= 2;
    }
    if (len) {
        sum += (uint32_t)(d[0] << 8);
    }
    while (sum >> 16) {
        sum = (sum & 0xffff) + (sum >> 16);
    }
    return swap16((uint16_t)~sum);
}
/* Minimal virtio-net driver over the virtio-mmio modern (v2) transport:
 * raw ethernet frames, no offloads.
 *
 * Adapted from sDDF's virtio network driver (github.com/au-ts/sddf,
 * BSD-2-Clause) and the OASIS VIRTIO 1.2 virtqueue layouts (BSD-3-Clause).
 */
#ifndef VIRTIO_NET_H
#define VIRTIO_NET_H

#include <stdbool.h>
#include <stdint.h>

/* Queue size for both RX and TX rings (power of two, <= QueueNumMax). */
#define VIRTIO_NET_QUEUE_SIZE 128
/* Per-frame buffer: 12-byte virtio-net header + frame, 2 KiB stride. */
#define VIRTIO_NET_BUF_SIZE 2048

/* Number of virtio-mmio slots in one 4 KiB window page (0x200 stride). */
#define VIRTIO_MMIO_SLOTS_PER_PAGE 8

/* Bring up the `nth` virtio-net device (counting from slot 0) in the
 * window page at `window_vaddr`, with its virtqueues in
 * [dma_vaddr, +2 MiB), uncached and physically contiguous, seen by the
 * device at dma_paddr.
 * Returns the slot index (0..7), negative on failure. */
int virtio_net_up(uintptr_t window_vaddr, unsigned nth, uintptr_t dma_vaddr, uintptr_t dma_paddr);

/* Pop one received frame, without its virtio-net header, into `dst`
 * (>= VIRTIO_NET_BUF_SIZE). False if nothing pending or the frame is empty. */
bool virtio_net_rx(uint8_t *dst, uint16_t *len);

/* Queue one frame for transmission; `frame` is copied. */
void virtio_net_tx(const uint8_t *frame, uint16_t len);

/* Read and acknowledge the device's interrupt status. */
uint32_t virtio_net_ack(void);

#endif /* VIRTIO_NET_H */
#include "virtio_net.h"

static inline void barrier(void)
{
    __sync_synchronize();
}

static int scan(uintptr_t window_vaddr, unsigned nth)
{
    unsigned seen = 0;
    for (int slot = 0; slot < VIRTIO_MMIO_SLOTS_PER_PAGE; slot++) {
        virtio_mmio_regs_t *r = (virtio_mmio_regs_t *)(window_vaddr + 0x200u * slot);
        if (r->MagicValue == VIRTIO_MMIO_MAGIC && r->DeviceID == VIRTIO_DEVICE_ID_NET) {
            if (seen == nth) {
                return slot;
            }
            seen++;
        }
    }
    return -1;
}

/* Lay out one queue at DMA offset `off`, each ring on its own 4 KiB
 * page; returns the offset past it. */
static uintptr_t vq_carve(struct vq *q, uintptr_t dma_vaddr, uintptr_t dma_paddr, uintptr_t off)
{
    uintptr_t desc_off = off;
    uintptr_t avail_off = desc_off + 0x1000;
    uintptr_t used_off = avail_off + 0x1000;
    uintptr_t bufs_off = used_off + 0x1000;

    q->desc = (volatile struct virtq_desc *)(dma_vaddr + desc_off);
    q->avail = (volatile struct virtq_avail *)(dma_vaddr + avail_off);
    q->used = (volatile struct virtq_used *)(dma_vaddr + used_off);
    q->bufs = (volatile uint8_t *)(dma_vaddr + bufs_off);
    q->bufs_paddr = dma_paddr + bufs_off;
    q->last_used = 0;

    for (unsigned i = 0; i < Q; i++) {
        q->desc[i].addr = q->bufs_paddr + (uint64_t)i * VIRTIO_NET_BUF_SIZE;
        q->desc[i].len = VIRTIO_NET_BUF_SIZE;
        q->desc[i].flags = 0;
        q->desc[i].next = 0;
    }
    q->avail->flags = 0;
    q->avail->idx = 0;
    q->used->flags = 0;

    return bufs_off + (uintptr_t)Q * VIRTIO_NET_BUF_SIZE;
}

static void vq_install(unsigned sel, const struct vq *q, uintptr_t dma_vaddr, uintptr_t dma_paddr)
{
    uint64_t desc = (uintptr_t)q->desc - dma_vaddr + dma_paddr;
    uint64_t avail = (uintptr_t)q->avail - dma_vaddr + dma_paddr;
    uint64_t used = (uintptr_t)q->used - dma_vaddr + dma_paddr;

    regs->QueueSel = sel;
    if (regs->QueueNumMax < Q) {
        return; /* caller checks QueueReady */
    }
    regs->QueueNum = Q;
    regs->QueueDescLow = (uint32_t)desc;
    regs->QueueDescHigh = (uint32_t)(desc >> 32);
    regs->QueueDriverLow = (uint32_t)avail;
    regs->QueueDriverHigh = (uint32_t)(avail >> 32);
    regs->QueueDeviceLow = (uint32_t)used;
    regs->QueueDeviceHigh = (uint32_t)(used >> 32);
    barrier();
    regs->QueueReady = 1;
}

static int virtio_net_init(uintptr_t regs_vaddr, uintptr_t dma_vaddr, uintptr_t dma_paddr)
{
    regs = (virtio_mmio_regs_t *)regs_vaddr;

    if (regs->MagicValue != VIRTIO_MMIO_MAGIC) {
        return -1;
    }
    if (regs->Version != 2) {
        return -2; /* legacy transport: launch QEMU with force-legacy=false */
    }
    if (regs->DeviceID != VIRTIO_DEVICE_ID_NET) {
        return -3;
    }

    /* VIRTIO 1.2 §3.1.1 initialisation sequence. */
    regs->Status = 0;
    regs->Status = STATUS_ACKNOWLEDGE;
    regs->Status = STATUS_ACKNOWLEDGE | STATUS_DRIVER;

    /* VERSION_1 only: no offloads. */
    regs->DriverFeaturesSel = 0;
    regs->DriverFeatures = 0;
    regs->DriverFeaturesSel = 1;
    regs->DriverFeatures = 1u << (VIRTIO_F_VERSION_1 - 32);
    regs->Status = STATUS_ACKNOWLEDGE | STATUS_DRIVER | STATUS_FEATURES_OK;
    if (!(regs->Status & STATUS_FEATURES_OK)) {
        regs->Status = STATUS_FAILED;
        return -4;
    }

    uintptr_t off = vq_carve(&rxq, dma_vaddr, dma_paddr, 0);
    (void)vq_carve(&txq, dma_vaddr, dma_paddr, off);
    tx_next = 0;

    vq_install(0, &rxq, dma_vaddr, dma_paddr);
    vq_install(1, &txq, dma_vaddr, dma_paddr);
    regs->QueueSel = 0;
    if (!regs->QueueReady) {
        regs->Status = STATUS_FAILED;
        return -5;
    }

    for (unsigned i = 0; i < Q; i++) {
        rxq.desc[i].flags = VIRTQ_DESC_F_WRITE;
        rxq.avail->ring[i] = (uint16_t)i;
    }
    barrier();
    rxq.avail->idx = Q;
    barrier();

    regs->Status = STATUS_ACKNOWLEDGE | STATUS_DRIVER | STATUS_FEATURES_OK | STATUS_DRIVER_OK;
    regs->QueueNotify = 0; /* kick RX so the device sees the buffers */
    return 0;
}

int virtio_net_up(uintptr_t window_vaddr, unsigned nth, uintptr_t dma_vaddr, uintptr_t dma_paddr)
{
    int slot = scan(window_vaddr, nth);
    if (slot < 0) {
        return -100;
    }
    int rc = virtio_net_init(window_vaddr + 0x200u * (unsigned)slot, dma_vaddr, dma_paddr);
    return rc == 0 ? slot : rc;
}

bool virtio_net_rx(uint8_t *dst, uint16_t *len)
{
    barrier();
    if (rxq.last_used == rxq.used->idx) {
        return false;
    }
    volatile struct virtq_used_elem *e = &rxq.used->ring[rxq.last_used % Q];
    uint16_t id = (uint16_t)e->id;
    uint32_t total = e->len;

    uint16_t frame_len = 0;
    if (total > NET_HDR_LEN) {
        frame_len = (uint16_t)(total - NET_HDR_LEN);
        volatile const uint8_t *src = rxq.bufs + (uint32_t)id * VIRTIO_NET_BUF_SIZE + NET_HDR_LEN;
        for (uint16_t i = 0; i < frame_len; i++) {
            dst[i] = src[i];
        }
    }
    *len = frame_len;
    rxq.last_used++;

    rxq.avail->ring[rxq.avail->idx % Q] = id;
    barrier();
    rxq.avail->idx++;
    barrier();
    regs->QueueNotify = 0;
    return frame_len != 0;
}

void virtio_net_tx(const uint8_t *frame, uint16_t len)
{
    uint16_t id = tx_next % Q;
    tx_next++;

    volatile uint8_t *buf = txq.bufs + (uint32_t)id * VIRTIO_NET_BUF_SIZE;
    for (unsigned i = 0; i < NET_HDR_LEN; i++) {
        buf[i] = 0; /* no offloads: zeroed virtio-net header */
    }
    for (uint16_t i = 0; i < len; i++) {
        buf[NET_HDR_LEN + i] = frame[i];
    }
    txq.desc[id].len = NET_HDR_LEN + len;
    txq.desc[id].flags = 0;

    txq.avail->ring[txq.avail->idx % Q] = id;
    barrier();
    txq.avail->idx++;
    barrier();
    regs->QueueNotify = 1;
}

uint32_t virtio_net_ack(void)
{
    uint32_t status = regs->InterruptStatus;
    regs->InterruptACK = status;
    return status;
}
/* Queries for `fire verify demo/datadiode/`. Each buffered channel is a
   queue of its length, the signal word and Microkit's pending bit, at most
   3 long in the model; the Environment raises each NIC's IRQ for ever, as
   inject.py keeps feeding frames, masked until the driver acks it, as seL4
   does (#355, #601). */

/* Never stuck: the device always has a move unless a driver never takes or
   never acks its IRQ, so a deadlock would be a stuck driver. */
A[] not deadlock

/* A frame crosses each hop: dd takes one from o2d, eth_inner one from d2i. */
E<> dd.L2
E<> eth_inner.Deliver

/* No lost wake-up: a parked receiver with a value waiting is always
   notified. A receiver stuck this way is not a deadlock, since eth_outer
   keeps running; these two catch it. */
A[] not (!dd.awake && q_o2d.length > 0 && !pend_o2d)
A[] not (!eth_inner.awake && q_d2i.length > 0 && !pend_d2i)

/* Secrecy: no Low PD, and nothing in o2d, ever holds a frame that came in
   on the inner (High) network. Frames carry the NIC they came in on, set by
   eth_frame's contract and proved on eth.c. */
A[] not (eth_outer.frame.origin == ORIGIN_INNER || dd.f.origin == ORIGIN_INNER || (exists (i : int[0, 2]) q_o2d_val[i].origin == ORIGIN_INNER))

/* The origin travels: dd receives a frame tagged with the outer network. */
E<> dd.f.origin == ORIGIN_OUTER
